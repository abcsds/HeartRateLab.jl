% ---------- z-score forest plot ----------
% \zrow{ypos}{z}{dotcolor}{feature name}
\newcommand{\zrow}[4]{%
  \draw[gray!45,line width=1.6pt] (0,#2) -- (#3,#2);
  \shade[ball color=#4] (#3,#2) circle (9pt);
  \node[anchor=east,text=white,font=\bfseries] at (-3.05,#2) {#1};
  \node[anchor=west,text=white,font=\bfseries] at (3.65,#2)
        {\pgfmathprintnumber[print sign,fixed,fixed zerofill,precision=2]{#3}$\sigma$};
}
% \zforest{max-y}{rows}  — draws bands + axis then the rows
\newcommand{\zforest}[2]{%
\begin{tikzpicture}[x=2.55cm,y=1.14cm]
  \def\ymax{#1}
  % dispersion bands
  \fill[jublue!22]  (-1,-0.55) rectangle (1,\ymax+0.55);
  \fill[zgold!26]   (-2,-0.55) rectangle (-1,\ymax+0.55);
  \fill[zgold!26]   (1,-0.55)  rectangle (2,\ymax+0.55);
  \fill[jured!16]   (-3,-0.55) rectangle (-2,\ymax+0.55);
  \fill[jured!16]   (2,-0.55)  rectangle (3.5,\ymax+0.55);
  % zero line + ticks
  \draw[white!85,dashed,line width=1.2pt] (0,-0.55) -- (0,\ymax+0.55);
  \foreach \t in {-2,-1,0,1,2,3}{
     \draw[white!45,line width=0.8pt] (\t,-0.55) -- (\t,-0.75);
     \node[below,text=white!80,font=\small] at (\t,-0.75) {\t};}
  \node[below,text=white!70,font=\small\itshape] at (0.5,-1.35)
        {z-equivalent (\,$\sigma$ from healthy-population median\,)};
  #2
\end{tikzpicture}}

% ================= ABSTRACT + DESCRIPTION =================
\begin{tcbraster}[raster columns=2, raster column skip=8mm, raster equal height=rows]
  \begin{tcolorbox}[card]
     \ctitle{Abstract}
     {\large
     A common method in the evaluation of human biosignals is comparing experimental
     signals against datasets of normative values. Thanks to the open scientific
     resources for heart rate analysis, and the resilient network of scientific
     programming functionality that the Julia community enables, the normative
     analysis of Heart Rate Variability (HRV) can be done \textbf{completely in Julia}.
     By contrasting features in windows of \textbf{360 heart beats} against normative
     datasets we identified that meditators from the \emph{Heart Rate Oscillations
     during Meditation} dataset (PhysioNet) presented indicators of increased
     \textbf{Mean IBI} (z\,$=\!+1.02$) and \textbf{pNN50} (z\,$=\!+1.32$), with
     smaller but consistent increases in other features common in
     resonant-frequency training that remained under 1$\sigma$:
     \textbf{RMSSD} (z\,$=\!+0.94$), \textbf{SDNN} (z\,$=\!+0.90$). Results
     demonstrate how correctly implemented open science can answer contextual
     questions such as: \emph{``Is this new experimental data normal?''}}
  \end{tcolorbox}
  \begin{tcolorbox}[card, borderline west={5mm}{0mm}{jublue}]
     \ctitle[jublue]{Why context matters}
     {\large
     We contrast a time series of \textbf{Inter-Beat Intervals (IBIs)} against the
     feature distributions of existing open datasets, emphasising the
     \textbf{contextual nature} of scientific inquiry. This capability is a novel
     addition implemented specifically for HeartRateLab.jl.\par\vspace{4mm}
     An example contrasting meditation recordings demonstrates current capabilities
     and illustrates functionality that aids \textbf{hypothesis testing},
     \textbf{feature engineering}, and a deeper understanding of recordings ---
     including use as a \textbf{personal daily HRV monitor} that norms your own
     signals against a healthy population.}
  \end{tcolorbox}
\end{tcbraster}

% ================= METHODS =================
\begin{tcolorbox}[card, borderline west={5mm}{0mm}{jupurple}]
  \ctitle[jupurple]{Method — a normative pipeline, end to end in Julia}
  \begin{minipage}[t]{0.70\linewidth}\large\vspace{-2mm}
   \textbf{\color{jured}1.}~Read IBIs \small(\texttt{read\_txt} / \texttt{read\_wfdb} /
   \texttt{read\_xdf}).\normalsize\\[2mm]
   \textbf{\color{jured}2.}~Preprocess: replace bio-outliers, ectopic \& zero beats;
   interpolate \texttt{NaN}s.\\[2mm]
   \textbf{\color{jured}3.}~Extract \textbf{53 HRV features} (time / frequency /
   nonlinear) via the \texttt{@register}\,+\,\texttt{@memoize} registry.\\[2mm]
   \textbf{\color{jured}4.}~\textbf{Window} each recording: 360 beats, 120-beat stride.\\[2mm]
   \textbf{\color{jured}5.}~Fit a \textbf{normative prior} per feature
   (Normal / Gamma / Beta / LogNormal) by MLE over \textbf{56{,}472 windows from
   66 healthy subjects} (PhysioNet \texttt{nsrdb}\,+\,\texttt{nsr2db}).\\[2mm]
   \textbf{\color{jured}6.}~Score any new window:
   $\ \text{percentile}=F_{\text{prior}}(x),\quad
   z_{\text{equiv}}=\Phi^{-1}(\text{percentile}).$
   \par\vspace{4mm}
   {\itshape\color{jcpurple} 68\,/\,95\,/\,99.7\% dispersion bands below come from the
   fitted priors --- \textmd{these are \textbf{window-level} (windows within a
   recording are correlated), so a $z$ is descriptive, not a per-subject
   inferential test.}}
  \end{minipage}\hfill
  \begin{minipage}[t]{0.275\linewidth}\vspace{-3mm}
    \centering
    \includegraphics[height=118mm]{figs/between_dataset_kde.png}\\[1mm]
    {\footnotesize\color{inkdark}\textbf{Validity check:} the two healthy reference
    databases (\texttt{nsrdb}, \texttt{nsr2db}) agree feature-by-feature, so the
    pooled prior is well-defined.}
  \end{minipage}
\end{tcolorbox}

% ================= RESULTS: two z-forests =================
\begin{tcbraster}[raster columns=2, raster column skip=8mm, raster equal height=rows]
  % ---- Case A: meditation cohort ----
  \begin{tcolorbox}[darkcard, colback=cardnavy, borderline west={5mm}{0mm}{jcgreen}]
     \ctitle{Case A — Meditation cohort}
     {\color{white!85}\large PhysioNet \emph{Heart Rate Oscillations during
     Meditation} · 58 records · median window vs healthy prior.}\par\vspace{6mm}
     \centering
     \zforest{3}{
        \zrow{pNN50}{3}{1.32}{zgold}
        \zrow{Mean IBI}{2}{1.02}{zgold}
        \zrow{RMSSD}{1}{0.94}{jcgreen}
        \zrow{SDNN}{0}{0.90}{jcgreen}
     }
     \par\vspace{5mm}
     {\raggedright\color{white!85}\large Meditators trend to \textbf{longer IBIs} and
     \textbf{elevated pNN50}; short-term vagal tone (RMSSD, SDNN) is
     \textbf{modestly elevated but under 1$\sigma$} --- a mild, coherent shift, not
     pathology.\par}
  \end{tcolorbox}
  % ---- Case B: anonymous longitudinal participant ----
  \begin{tcolorbox}[darkcard, colback=cardnavy, borderline west={5mm}{0mm}{jured}]
     \ctitle[jured]{Case B — Anonymous longitudinal participant}
     {\color{white!85}\large A \textbf{daily HRV monitor}: 148 windows across
     \textbf{88 recording dates}, 2019--2025, vs the same healthy prior.}\par\vspace{5mm}
     \centering
     \zforest{8}{
        \zrow{LF/HF ratio}{8}{2.64}{jured}
        \zrow{LF Power}{7}{2.55}{jured}
        \zrow{SD2}{6}{1.19}{zgold}
        \zrow{SDNN}{5}{1.12}{zgold}
        \zrow{pNN50}{4}{0.74}{jcgreen}
        \zrow{Mean IBI}{3}{0.07}{jcgreen}
        \zrow{SD1}{2}{0.03}{jcgreen}
        \zrow{RMSSD}{1}{0.03}{jcgreen}
        \zrow{HF Power}{0}{-0.17}{jcgreen}
     }
     \par\vspace{4mm}
     {\raggedright\color{white!90}\large\bfseries The resonance signature:
     \normalfont\color{white!85} the $\sim$0.1\,Hz \textbf{LF band} is strongly elevated
     (LF power \& the LF/HF ratio derived from it, $>\!+2.5\sigma$) while parasympathetic
     \& short-term metrics (RMSSD, HF, SD1, Mean IBI) stay near the healthy median. The
     monitor's $z$-scores \emph{track the practice} --- consistent with slow, resonant
     breathing.\par\vspace{2mm}
     {\footnotesize\color{white!70} Single participant; 148 correlated windows
     (descriptive, not inferential). Features are not independent
     --- SD1\,$\equiv$\,RMSSD$/\sqrt2$, SD2 tracks SDNN, LF/HF derives from LF power
     --- so nine labels span $\sim$5 axes.}\par}
  \end{tcolorbox}
\end{tcbraster}
